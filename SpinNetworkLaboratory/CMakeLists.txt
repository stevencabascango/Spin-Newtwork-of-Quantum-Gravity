cmake_minimum_required(VERSION 3.16)
project(SpinNetworkLaboratory VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SNL_BUILD_DESKTOP "Build the Workshop 1 SFML application" ON)
add_library(snl_core INTERFACE)
target_include_directories(snl_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(snl_core INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(spin_console src/console.cpp)
target_link_libraries(spin_console PRIVATE snl_core)
if(SNL_BUILD_DESKTOP)
  find_package(SFML 2.6 COMPONENTS graphics window system REQUIRED)
  add_executable(spin_network main.cpp src/Application/DesktopApplication.cpp
    src/Graphics/GraphRenderer.cpp src/Graphics/InputController.cpp src/Graphics/HUD.cpp)
  target_link_libraries(spin_network PRIVATE snl_core sfml-graphics sfml-window sfml-system)
  add_custom_command(TARGET spin_network POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:spin_network>/assets)
endif()
enable_testing()
foreach(suite core physics interaction)
  add_executable(test_${suite} tests/test_${suite}.cpp)
  target_link_libraries(test_${suite} PRIVATE snl_core)
  add_test(NAME ${suite} COMMAND test_${suite})
endforeach()
